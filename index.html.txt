// ==========================================
// 1. REGISTRATION SCREEN (Multi-Child & Photos)
// ==========================================
class BTOSRegistrationScreen extends StatefulWidget {
  const BTOSRegistrationScreen({Key? key}) : super(key: key);

  @override
  _BTOSRegistrationScreenState createState() => _BTOSRegistrationScreenState();
}

class _BTOSRegistrationScreenState extends State<BTOSRegistrationScreen> {
  final _formKey = GlobalKey<FormState>();
  
  String? selectedDistrict;
  int selectedChildrenCount = 1; 

  // දරුවන්ගේ දත්ත තබාගැනීමට List එකක් (නම, ශ්‍රේණිය, ඡායාරූපය)
  final List<Map<String, dynamic>> _childrenData = [
    {'nameController': TextEditingController(), 'grade': 1, 'imageFile': null}
  ];

  final List<String> districts = [
    'Anuradhapura', 'Colombo', 'Gampaha', 'Kandy', 'Galle', 'Kurunegala', 'Ratnapura', 'Matale', 'Jaffna'
  ];

  // ඡායාරූපයක් ලබාගැනීම සඳහා Function එක (Camera හෝ Gallery)
  Future<void> _pickImage(int index, ImageSource source) async {
    final picker = ImagePicker();
    final pickedFile = await picker.pickImage(source: source);
    if (pickedFile != null) {
      setState(() {
        _childrenData[index]['imageFile'] = File(pickedFile.path);
      });
    }
  }

  // දරුවන්ගේ ගණන වෙනස් වන විට List එක Update කිරීම
  void _updateChildrenCount(int count) {
    setState(() {
      selectedChildrenCount = count;
      if (_childrenData.length < count) {
        for (int i = _childrenData.length; i < count; i++) {
          _childrenData.add({
            'nameController': TextEditingController(),
            'grade': 1,
            'imageFile': null,
          });
        }
      } else if (_childrenData.length > count) {
        _childrenData.removeRange(count, _childrenData.length);
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text("BTOS - Student Registration", style: TextStyle(fontWeight: FontWeight.bold)),
        backgroundColor: Colors.lightBlue,
      ),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20),
        child: Form(
          key: _formKey,
          child: Column(
            crossAxisAlignment: Cross.stretch,
            children: [
              const Text(
                "ආයුබෝවන්! කරුණාකර ඔබගේ ගිණුම සක්‍රීය කරන්න.",
                style: TextStyle(fontSize: 16, fontWeight: FontWeight.bold, color: Colors.orange),
                textAlign: TextAlign.center,
              ),
              const SizedBox(height: 20),
              
              // WhatsApp Number Field
              TextFormField(
                keyboardType: TextInputType.phone,
                decoration: const InputDecoration(
                  labelText: "WhatsApp Number",
                  border: OutlineInputBorder(),
                  prefixIcon: Icon(Icons.phone),
                ),
              ),
              const SizedBox(height: 15),

              // District Dropdown
              DropdownButtonFormField<String>(
                decoration: const InputDecoration(labelText: "District (දිස්ත්‍රික්කය)", border: OutlineInputBorder()),
                value: selectedDistrict,
                items: districts.map((district) {
                  return DropdownMenuItem(value: district, child: Text(district));
                }).toList(),
                onChanged: (val) => setState(() => selectedDistrict = val),
              ),
              const SizedBox(height: 20),

              // Number of Children Dropdown (1 to 5)
              DropdownButtonFormField<int>(
                decoration: const InputDecoration(
                  labelText: "ඉගෙන ගන්නා දරුවන් ප්‍රමාණය (Number of Children)", 
                  border: OutlineInputBorder(),
                ),
                value: selectedChildrenCount,
                items: [1, 2, 3, 4, 5].map((count) {
                  return DropdownMenuItem(value: count, child: Text("$count දරුවන් (Children)"));
                }).toList(),
                onChanged: (val) {
                  if (val != null) _updateChildrenCount(val);
                },
              ),
              const SizedBox(height: 20),

              // Dynamic Children Forms
              ListView.builder(
                shrinkWrap: true,
                physics: const NeverScrollableScrollPhysics(),
                itemCount: selectedChildrenCount,
                itemBuilder: (context, index) {
                  return Card(
                    margin: const EdgeInsets.symmetric(vertical: 8),
                    elevation: 3,
                    child: Padding(
                      padding: const EdgeInsets.all(12),
                      child: Column(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          Text("දරුවා ${index + 1} විස්තර (Child ${index + 1})", 
                            style: const TextStyle(fontWeight: FontWeight.bold, color: Colors.blue, fontSize: 16)),
                          const SizedBox(height: 10),
                          
                          // Child Name
                          TextFormField(
                            controller: _childrenData[index]['nameController'],
                            decoration: InputDecoration(
                              labelText: "දරුවා ${index + 1} ගේ නම", 
                              border: const OutlineInputBorder()
                            ),
                          ),
                          const SizedBox(height: 10),

                          // Grade Selection (Grade 1 to 5)
                          DropdownButtonFormField<int>(
                            decoration: const InputDecoration(labelText: "ශ්‍රේණිය (Grade 1 - 5)", border: OutlineInputBorder()),
                            value: _childrenData[index]['grade'],
                            items: [1, 2, 3, 4, 5].map((g) => DropdownMenuItem(value: g, child: Text("Grade $g"))).toList(),
                            onChanged: (val) {
                              setState(() {
                                _childrenData[index]['grade'] = val;
                              });
                            },
                          ),
                          const SizedBox(height: 10),

                          // Profile Photo Display & Upload Options
                          Row(
                            children: [
                              CircleAvatar(
                                radius: 30,
                                backgroundColor: Colors.grey[300],
                                backgroundImage: _childrenData[index]['imageFile'] != null
                                    ? FileImage(_childrenData[index]['imageFile'])
                                    : null,
                                child: _childrenData[index]['imageFile'] == null
                                    ? const Icon(Icons.person, size: 30, color: Colors.grey)
                                    : null,
                              ),
                              const SizedBox(width: 15),
                              Expanded(
                                child: Column(
                                  children: [
                                    ElevatedButton.icon(
                                      style: ElevatedButton.styleFrom(backgroundColor: Colors.teal),
                                      onPressed: () => _pickImage(index, ImageSource.camera),
                                      icon: const Icon(Icons.camera_alt, color: Colors.white, size: 18),
                                      label: const Text("කැමරාවෙන් Photo එකක් ගන්න", style: TextStyle(color: Colors.white, fontSize: 12)),
                                    ),
                                    const SizedBox(height: 5),
                                    OutlinedButton.icon(
                                      onPressed: () => _pickImage(index, ImageSource.gallery),
                                      icon: const Icon(Icons.photo_library, size: 18),
                                      label: const Text("ගැලරියෙන් Photo එකක් දාන්න", style: TextStyle(fontSize: 12)),
                                    ),
                                  ],
                                ),
                              ),
                            ],
                          ),
                        ],
                      ),
                    ),
                  );
                },
              ),
              const SizedBox(height: 20),

              // Activate Button
              ElevatedButton(
                style: ElevatedButton.styleFrom(
                  backgroundColor: Colors.lightBlue,
                  padding: const EdgeInsets.symmetric(vertical: 15),
                ),
                onPressed: () {
                  // Dashboard වෙත යොමු කිරීම
                  Navigator.pushReplacement(
                    context,
                    MaterialPageRoute(builder: (context) => const BTOSDashboardScreen()),
                  );
                },
                child: const Text("Activate Account & Login", style: TextStyle(fontSize: 18, color: Colors.white)),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
